cmake_minimum_required(VERSION 3.20)
project(mcumgr_consumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(MCUMGR_ROOT "" CACHE PATH "Extracted SDK directory (include/ and lib/)")
if(NOT MCUMGR_ROOT)
  message(FATAL_ERROR "Set MCUMGR_ROOT to the extracted SDK")
endif()
add_library(mcumgr_sdk STATIC IMPORTED)
if(WIN32 AND MSVC)
  set_target_properties(mcumgr_sdk PROPERTIES IMPORTED_LOCATION "${MCUMGR_ROOT}/lib/mcumgr.lib")
  # Rust MSVC target defaults to dynamic CRT (/MD, /MDd).
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  set_property(TARGET mcumgr_sdk PROPERTY INTERFACE_LINK_LIBRARIES
    "advapi32;bcrypt;crypt32;iphlpapi;kernel32;msvcrt;netapi32;ntdll;ole32;oleaut32;propsys;runtimeobject;setupapi;shell32;user32;userenv;uuid;version;windowsapp;winspool;ws2_32;secur32")
elseif(APPLE)
  set_target_properties(mcumgr_sdk PROPERTIES IMPORTED_LOCATION "${MCUMGR_ROOT}/lib/libmcumgr.a")
  set_property(TARGET mcumgr_sdk PROPERTY INTERFACE_LINK_LIBRARIES
    "-framework Foundation;-framework CoreBluetooth;-framework IOKit;-framework Security;-framework SystemConfiguration;-lobjc;-lc++")
else()
  message(FATAL_ERROR "Supported SDKs: Windows MSVC x64 or macOS Intel")
endif()
set_target_properties(mcumgr_sdk PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${MCUMGR_ROOT}/include")
enable_testing()
foreach(kind IN ITEMS c cpp)
  add_executable(example_${kind} example.${kind})
  target_link_libraries(example_${kind} PRIVATE mcumgr_sdk)
  add_test(NAME example_${kind} COMMAND example_${kind})
endforeach()
